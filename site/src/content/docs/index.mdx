## Quick start

```ini
# postgresql.conf (one restart)
shared_preload_libraries     = 'pgbx'
pgbx.s3_endpoint      = 'https://hel1.your-objectstorage.com'
pgbx.s3_bucket        = 'my-backups'
pgbx.s3_region        = 'hel1'
pgbx.server_name      = 'db-prod-1'
pgbx.credentials_file = '/etc/pgbx/s3.credentials'
```

```sql
CREATE DATABASE myapp;                          -- backed up within a minute
\c myapp
SELECT * FROM pgbx.status();             -- state, schedule, last + next backup
SELECT pgbx.backup_now();                -- returns a job id
SELECT pgbx.restore(into_db => 'myapp_restored', at => '2026-10-01 09:00');
```
